## 自己封装的函数
```c

size_t readn(int fd, void *buf, size_t n){
  
 left=n;

 while(left>0){
     nread= read(fd,buf,left)
    if(nread<0){
        if(errno==EINTR){ //非阻塞中断
            nread=0;
            continue;
        }else{ 
            return -1;
        }

    } if(nread==0){
         //关闭
        break;
    } else{
         left-=nread;
        buf+=nread;
    }
 }

 return n-left;
 
}



size_t readline(int fd,void *buf,size_t maxlen,int *close){
    size_t n,rc;
    char c;
    for(n=0;n<maxlen;n++){
        rc=read(fd,&c,1);
        // 判断是否是FIN 关闭
        if(rc==0) {
            *close=1;
            break;
        }
        if(rc>0){
            buf[n]=c;
            // 读取到换行
            if(c=='\n')  break;
        }else{
            if(errno==EINTR){
                continue;
            }
            // 缓冲区没有数据了
            if(errno==EAGAIN || errno==EWOULDBLOCK){
                break;
            }
            return -1;
        }
    }

    return n;

}




size_t writen(int fd,const void *buf,size_t n,int *close){

    size_t nwrite=0;
    while(nwrite<n){
        int n=write(fd,buf+nwrite,n-nwrite);
        if(n<=0){
            if(errno==EINTR){
                continue;
            }else if(errno==EPIPE){
                *close=1;
                break;
            } else {
              break;
            }
        }else{
            nwrite+=n;
        }
    }

    return nwrite;

}


```
