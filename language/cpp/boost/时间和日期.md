## timer
```cpp

#include <iostream>
#include <boost/timer.hpp>

int main()
{
    boost::timer t; // 声明一个计时器对象开始计数
    // 可以度量的最大时间  以小时为单位
    std::cout << "max timespan:" << t.elapsed_max() / 3600 << std::endl;
    // 可以度量的最小时间  以秒为单位
    std::cout << "min timespan:" << t.elapsed_min() << std::endl;
    
    // 已经流逝的时间
    std::cout << "now time elapsed:" << t.elapsed() << std::endl;
}


// 自进程启动已经的clock 数
std::clock();




```
