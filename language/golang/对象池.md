# 对象池
```golang


package main

import (
	"fmt"
	"strconv"
	"sync"
)

type IObj interface {
	Print()
}

type Obj struct {
	Name string
}

func (self *Obj) Print() {
	fmt.Println(self.Name)
}


// 生成新对象函数
var i = 0
var pool sync.Pool = sync.Pool{New: func() any {
    i = i + 1
	fmt.Println("New i:", i)
	return &Obj{Name: strconv.Itoa(i)}
}}


// 获取对象
func GetObj() *Obj {
	return pool.Get().(*Obj)
}

// 插入对象到池

func PutObj(obj *Obj) {
	fmt.Printf("release Name:%s\n", obj.Name)
	pool.Put(obj)
}

func main() {

	objs := make([]*Obj, 0)

	for i := 0; i < 10; i++ {
		objs = append(objs, GetObj())
	}

	for _, o := range objs {
		o.Print()
		PutObj(o)
	}

}


//对象池的基本使用


1.该对象池并发安全、


2.pool.put 并不保证数据都存入池里面 可能会丢弃该对象


3.put(obj)  get 出来的不一定是刚才put 进去的对象
```
