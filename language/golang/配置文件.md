# 配置文件
## ini 文件解析
```golang
go get github.com/unknwon/goconfig


package main

import (
	"fmt"
	"log"

	"github.com/unknwon/goconfig"
)

func main() {
	cfg, err := goconfig.LoadConfigFile("./test.ini")

	if err != nil {
		log.Fatal(err)
	}

	// 读取默认section 值
	google, _ := cfg.GetValue("", "google")
	search, _ := cfg.GetValue(goconfig.DEFAULT_SECTION, "search")
	fmt.Println("google:", google)
	fmt.Println("search:", search)

	cfg.SetValue("parent", "name", "crl")
	parent, _ := cfg.GetSection("parent")

	fmt.Printf("parent section:%v\n", parent)

    // 写入数据
	err = goconfig.SaveConfigFile(cfg, "./test2.ini")
	if err != nil {
		log.Fatal(err)
	}

}


note:
[xxx] 表示 section 

如果没有[] 表示 default-section



```
